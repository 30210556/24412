// --- Google Apps Script / OpenAI 연동 및 자동 변환 기능 --- askAiBtn.addEventListener('click', async () => { const q = aiQuestion.value.trim(); if(!q) { alert('질문을 입력하세요.'); return; } if(GAS_WEB_APP_URL.includes('YOUR_GOOGLE')) { alert('Google Apps Script 웹 앱 URL 주소를 입력해 주세요.'); return; } try { // 버튼 비활성화 및 상태 표시 askAiBtn.disabled = true; ttsBtn.disabled = true; aiAnswer.textContent = "AI 답변 생성 중입니다. 잠시만 기다려주세요..."; setStatus('Apps Script API와 통신 중...'); // 1. ChatGPT API 호출 (Apps Script 경유) const res = await fetch(GAS_WEB_APP_URL, { method: 'POST', body: JSON.stringify({ type: "chat", question: q }) }); const data = await res.json(); if (data.answer) { const reply = data.answer; lastAiAnswerText = reply; aiAnswer.textContent = reply; setStatus('답변 수신 완료! 수학 기호 변환 및 TTS를 준비합니다.'); // 🌟 [핵심 연동] AI 답변을 수학 변환기 입력창에 자동으로 주입 mathInput.value = reply; // 🌟 수학 변환 함수를 강제로 실행시켜 결과창에 기호(ex: ∫, √)가 뜨게 만듦 convertMathText(); // 🌟 답변을 받자마자 사용자가 버튼을 누르지 않아도 자동으로 TTS 재생 시작 playAutomaticTTS(reply); } else if (data.error) { aiAnswer.textContent = `오류 발생:\n${data.error}`; setStatus('API 에러 응답 수신'); askAiBtn.disabled = false; } } catch (err) { console.error(err); aiAnswer.textContent = "통신 실패 오류가 발생했습니다. URL 또는 CORS 세팅을 확인하세요."; setStatus('서버 통신 실패'); askAiBtn.disabled = false; } }); // 수동으로 다시 듣고 싶을 때 사용하는 버튼 이벤트 ttsBtn.addEventListener('click', () => { if(lastAiAnswerText) playAutomaticTTS(lastAiAnswerText); }); // 🔊 TTS 자동 재생 처리 함수 async function playAutomaticTTS(textToSpeak) { try { ttsBtn.disabled = true; setStatus('AI 답변을 음성으로 변환 중입니다...'); // 대량의 텍스트 오버플로우 방지를 위해 앞부분 200자만 안전하게 슬라이싱 const cleanText = textToSpeak.substring(0, 200); const res = await fetch(GAS_WEB_APP_URL, { method: 'POST', body: JSON.stringify({ type: "tts", text: cleanText }) }); const data = await res.json(); if (data.audioBase64) { setStatus('🔊 AI 답변 음성 재생 중...'); const audioBytes = Uint8Array.from(atob(data.audioBase64), c => c.charCodeAt(0)); const blob = new Blob([audioBytes], { type: 'audio/mp3' }); const audioUrl = URL.createObjectURL(blob); const audio = new Audio(audioUrl); audio.play(); // 음성 재생이 완전히 끝나면 다시 질문 버튼 활성화 audio.onended = () => { setStatus('음성 안내가 완료되었습니다.'); ttsBtn.disabled = false; askAiBtn.disabled = false; }; } else { setStatus('TTS 변환 실패 (오디오 데이터 없음)'); ttsBtn.disabled = false; askAiBtn.disabled = false; } } catch (err) { console.error(err); setStatus('TTS 재생 중 통신 오류 발생'); ttsBtn.disabled = false; askAiBtn.disabled = false; } }
